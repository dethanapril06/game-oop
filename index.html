                            1. Blueprint class Robot:
                            Cetakan Objek
                        

                        

class Robot:
    # Class Attribute (Milik Bersama)
    pabrik = "CyberNusantara"

    # Constructor (__init__)
    def __init__(self, nama, tipe, energi):
        self.nama = nama
        self.tipe = tipe
        self.energi = energi

    # Method (Perilaku Objek)
    def isi_daya(self):
        self.energi += 20
